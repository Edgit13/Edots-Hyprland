#!/bin/bash
# Legacy note kept only as a reference for the old game mode idea.
# The active implementation is now quickshell/scripts/gamemode.sh
# and is compositor-agnostic for the MangoWM setup.

STATE_DIR="$HOME/.cache/quickshell"
STATE_FILE="$STATE_DIR/gamemode.state"
mkdir -p "$STATE_DIR"

current_state="0"
if [ -f "$STATE_FILE" ]; then
    current_state=$(tr -d ' \n\r\t' < "$STATE_FILE" 2>/dev/null)
fi

if [ "$current_state" = "1" ]; then
    swaync-client -df >/dev/null 2>&1 || true
    if command -v cpupower >/dev/null 2>&1; then
        pkexec cpupower frequency-set -g schedutil >/dev/null 2>&1 &
    fi
    notify-send -a "GameMode" "Game Mode: OFF" "Defaults restored" -i input-gaming >/dev/null 2>&1 || true
    printf '0\n' > "$STATE_FILE"
    exit 0
fi

swaync-client -dn >/dev/null 2>&1 || true
if command -v cpupower >/dev/null 2>&1; then
    pkexec cpupower frequency-set -g performance >/dev/null 2>&1 &
fi
notify-send -a "GameMode" "Game Mode: ON" "Notifications muted and performance mode requested" -i input-gaming >/dev/null 2>&1 || true
printf '1\n' > "$STATE_FILE"
