#!/bin/bash
STATE_DIR="$HOME/.cache/quickshell"
STATE_FILE="$STATE_DIR/gamemode.state"
LOG_FILE="$STATE_DIR/gamemode.log"
mkdir -p "$STATE_DIR"

exec >> "$LOG_FILE" 2>&1
echo "--- $(date) ---"

current_state="0"
if [ -f "$STATE_FILE" ]; then
    current_state=$(tr -d ' \n\r\t' < "$STATE_FILE" 2>/dev/null)
fi

if [ "$current_state" = "1" ]; then
    hyprctl reload
    swaync-client -df >/dev/null 2>&1 || true
    if command -v cpupower >/dev/null 2>&1; then
        pkexec cpupower frequency-set -g schedutil >/dev/null 2>&1 &
    fi
    notify-send -a "GameMode" "Game Mode: OFF" "Defaults restored from Hyprland config" -i input-gaming >/dev/null 2>&1 || true
    printf '0\n' > "$STATE_FILE"
    exit 0
fi

hyprctl eval "hl.config({
    general = { gaps_in = 0, gaps_out = 0, border_size = 0 },
    animations = { enabled = 0 },
    decoration = { shadow = { enabled = 0 }, blur = { enabled = 0 }, rounding = 0 },
    misc = {},
})"

swaync-client -dn >/dev/null 2>&1 || true
if command -v cpupower >/dev/null 2>&1; then
    pkexec cpupower frequency-set -g performance >/dev/null 2>&1 &
fi
notify-send -a "GameMode" "Game Mode: ON" "Rounding, blur, shadow, gaps and borders disabled" -i input-gaming >/dev/null 2>&1 || true
printf '1\n' > "$STATE_FILE"
